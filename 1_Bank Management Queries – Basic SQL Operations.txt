Consider following Relation

	Customer(cust_name,cust_street,cust_city) 
	Depositor(cust_name,acc_no) 
	Loan(loan_no,branch_name,amount) 
	Borrower(cust_name,loan_no) 
Create above tables with appropriate constraints like primary key,foreign key, not null etc. 
1. Find the names of all branches in loan relation. 
2. Find all loan numbers for loans made at „Wadia College‟ Branch with loan amount > 12000. 
3. Find all customers who have a loan from bank. Find their names, loan_no and loan amount. 
4. List all customers in alphabetical order who have loan from „Wadia College‟ branch. 
5. Display distinct cities of branch.

CREATE DATABASE Practical;
USE Practical;



CREATE TABLE Branch (
    branch_name VARCHAR(50) PRIMARY KEY,
    branch_city VARCHAR(50) NOT NULL,
    asset DECIMAL(12,2)
);

INSERT INTO Branch VALUES

CREATE TABLE Account (
    acc_no INT PRIMARY KEY,
    branch_name VARCHAR(50),
    balance DECIMAL(12,2) CHECK (balance >= 0),
    FOREIGN KEY (branch_name)
        REFERENCES Branch(branch_name)
);

CREATE TABLE Customer (
    cust_name VARCHAR(50) PRIMARY KEY,
    cust_street VARCHAR(50),
    cust_city VARCHAR(50)
);

INSERT INTO Customer VALUES

('Sneha','JM Road','Pune'),

CREATE TABLE Depositor (
    cust_name VARCHAR(50),
    acc_no INT,
    PRIMARY KEY (cust_name, acc_no),
    FOREIGN KEY (cust_name)
        REFERENCES Customer(cust_name),
    FOREIGN KEY (acc_no)
        REFERENCES Account(acc_no)
);

INSERT INTO Depositor VALUES

('Amit',103),
('Priya',104);

CREATE TABLE Loan (
    loan_no INT PRIMARY KEY,
    branch_name VARCHAR(50),
    amount DECIMAL(12,2) CHECK (amount > 0),
    FOREIGN KEY (branch_name)
        REFERENCES Branch(branch_name)
);

INSERT INTO Loan VALUES

(202,'FC Road',10000),
(203,'Shivaji Nagar',25000);

CREATE TABLE Borrower (
    cust_name VARCHAR(50),
    loan_no INT,
    PRIMARY KEY (cust_name, loan_no),
    FOREIGN KEY (cust_name)
        REFERENCES Customer(cust_name),
    FOREIGN KEY (loan_no)
        REFERENCES Loan(loan_no)
);

SELECT branch_name
FROM Loan;

SELECT loan_no
FROM Loan
WHERE branch_name = 'Wadia College'
AND amount > 12000;


SELECT Borrower.cust_name, Loan.loan_no, Loan.amount
FROM Borrower
INNER JOIN Loan
ON Borrower.loan_no = Loan.loan_no;


SELECT Borrower.cust_name
FROM Borrower
INNER JOIN Loan
ON Borrower.loan_no = Loan.loan_no
WHERE Loan.branch_name = 'Wadia College'
ORDER BY Borrower.cust_name ASC;


SELECT DISTINCT branch_city
FROM Branch;
